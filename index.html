🏗️ Hausbau Nacharbeit

Einmalige Einrichtung mit Supabase — danach läuft die App auf allen Geräten.

✅ 100% kostenlos — Supabase Spark-Tarif: 500 MB Datenbank, 1 GB Storage, kein Upgrade nötig.
  1. Gehe zu supabase.com → Konto erstellen → New project
  2. Projektname eingeben (z.B. "Hausbau"), Passwort merken, Region wählen → Create project
  3. Links auf SQL Editor → New query → folgenden SQL einfügen & ausführen:
-- Tabellen erstellen create table if not exists etagen ( id uuid default gen_random_uuid() primary key, name text not null, plan_base64 text, created_at timestamptz default now() ); create table if not exists contacts ( id uuid default gen_random_uuid() primary key, name text not null, phone text, email text, firma text ); create table if not exists punkte ( id uuid default gen_random_uuid() primary key, nummer int, titel text not null, kategorie text not null default 'critical', beschreibung text, etage_id uuid references etagen(id), verantwortlicher_id uuid references contacts(id), due_date date, loc_x float8, loc_y float8, status text default 'open', foto_count int default 0, aufgenommen_von text, erstellt timestamptz default now() ); -- Migration (falls Tabelle bereits existiert): alter table public.punkte add column if not exists aufgenommen_von text; create table if not exists photos ( id uuid default gen_random_uuid() primary key, punkt_id uuid references punkte(id) on delete cascade, idx int, base64 text not null ); create table if not exists meta ( key text primary key, value bigint default 0 ); insert into meta (key, value) values ('counter', 0) on conflict do nothing; -- Berechtigungen für anon-Rolle (App nutzt anon key) grant select, insert, update, delete on public.etagen to anon, authenticated, service_role; grant select, insert, update, delete on public.contacts to anon, authenticated, service_role; grant select, insert, update, delete on public.punkte to anon, authenticated, service_role; grant select, insert, update, delete on public.photos to anon, authenticated, service_role; grant select, insert, update, delete on public.meta to anon, authenticated, service_role; -- RLS aktivieren alter table public.etagen enable row level security; alter table public.contacts enable row level security; alter table public.punkte enable row level security; alter table public.photos enable row level security; alter table public.meta enable row level security; -- Policies: anon darf alles (App hat eigenen Login-Schutz) create policy "anon_full" on public.etagen for all to anon using (true) with check (true); create policy "anon_full" on public.contacts for all to anon using (true) with check (true); create policy "anon_full" on public.punkte for all to anon using (true) with check (true); create policy "anon_full" on public.photos for all to anon using (true) with check (true); create policy "anon_full" on public.meta for all to anon using (true) with check (true);
